cmake_minimum_required(VERSION 3.16)
project(native_stage_ninth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenFHE CONFIG REQUIRED)
set(APP "" CACHE PATH "Pinned app checkout")
add_executable(stage_ninth src/stage.cpp src/FHEController.cpp)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
target_include_directories(stage_ninth PRIVATE "${APP}/src" ${OPENMP_INCLUDES} "${OpenFHE_INCLUDE}" "${OpenFHE_INCLUDE}/third-party/include" "${OpenFHE_INCLUDE}/core" "${OpenFHE_INCLUDE}/pke" "${OpenFHE_INCLUDE}/binfhe")
target_link_directories(stage_ninth PRIVATE "${OpenFHE_LIBDIR}")
target_link_libraries(stage_ninth PRIVATE ${OpenFHE_SHARED_LIBRARIES})
